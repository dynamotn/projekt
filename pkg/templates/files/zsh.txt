# vim:filetype=zsh
pj() {
  if (( $# > 1 )); then
    print -u2 "usage: pj [project]"
    return 1
  fi

  local target
  # `projekt folder get` exits non-zero when the short name is unknown, so the
  # error it already printed is the only message the user needs. With no
  # argument there is nothing to resolve, so ask instead.
  if (( $# == 0 )); then
    target=$(command projekt folder select --verbose=error) || return 1
  else
    target=$(command projekt folder get --verbose=error -- "$1") || return 1
  fi

  if [[ ! -d $target ]]; then
    print -u2 "Project folder does not exist: $target"
    return 1
  fi

  cd -- "$target" || return 1

  # The argument is not the name we landed on when it was "-", a loose match
  # or nothing at all, so the tab is named after the folder itself.
  local label=${target:t}
  [[ -n ${ZELLIJ:-} ]] && zellij action rename-tab "$label"
  [[ -n ${TMUX:-} ]] && tmux rename-window "$label"
  return 0
}

_pj() {
  # A project is the only argument, so there is nothing to offer after it.
  (( CURRENT > 2 )) && return 1

  local -a lines described plain
  local line name description
  # --output tsv prints one project per line, so completion does not have to
  # strip table borders and cannot be broken by a name containing one. The
  # columns are the short name, the path, the workspace, the tags and the
  # description; a project with a description is offered with it beside it.
  lines=(${(f)"$(command projekt folder list --output tsv --no-headers --verbose=error 2>/dev/null)"})
  (( ${#lines} )) || return 1

  for line in $lines; do
    name=${line%%$'\t'*}
    description=${${(ps:\t:)line}[5]}
    if [[ -n $description ]]; then
      # A colon separates the candidate from its description for _describe,
      # so one inside the name has to be escaped.
      described+=("${name//:/\\:}:${description}")
    else
      plain+=("$name")
    fi
  done

  (( ${#described} )) && _describe -t projects 'project' described
  (( ${#plain} )) && compadd -a plain
  return 0
}

if (( $+functions[compdef] )); then
  compdef _pj pj
else
  # compinit has not run yet. Sourcing this near the top of a .zshrc is a
  # normal thing to do, so register the completion at the first prompt instead
  # of losing it.
  autoload -Uz add-zsh-hook
  __pj_register_completion() {
    (( $+functions[compdef] )) && compdef _pj pj
    add-zsh-hook -d precmd __pj_register_completion
    unfunction __pj_register_completion
  }
  add-zsh-hook precmd __pj_register_completion
fi
