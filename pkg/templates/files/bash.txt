#!/bin/bash
# vim:filetype=sh
function pj() {
  if [ $# -gt 1 ]; then
    echo "usage: pj [project]" >&2
    return 1
  fi

  local target
  # `projekt folder get` exits non-zero when the short name is unknown, so the
  # error it already printed is the only message the user needs. With no
  # argument there is nothing to resolve, so ask instead. The `--` keeps a name
  # starting with a dash from being read as a flag.
  if [ $# -eq 0 ]; then
    target=$(command projekt folder select --verbose=error) || return 1
  else
    target=$(command projekt folder get --verbose=error -- "$1") || return 1
  fi

  if [ ! -d "$target" ]; then
    echo "Project folder does not exist: $target" >&2
    return 1
  fi

  local from=$PWD
  cd -- "$target" || return 1

  # The hooks of the project left and the one entered, evaluated here because
  # that is the only place an `export` or a `nvm use` can take effect. A hook
  # that is refused is reported on stderr and the jump stands.
  eval "$(command projekt folder hooks --from "$from" --to "$target" --verbose=error)"

  # The argument is not the name we landed on when it was "-", a loose match
  # or nothing at all, so the tab is named after the folder itself.
  local label
  label=$(basename -- "$target")
  [ -n "$ZELLIJ" ] && zellij action rename-tab "$label"
  [ -n "$TMUX" ] && tmux rename-window "$label"
  return 0
}

function _pj_completion() {
  local cur=${COMP_WORDS[COMP_CWORD]} line
  COMPREPLY=()
  # The names are read one per line and matched here rather than through
  # `compgen -W`, which would expand each of them as shell code: a workspace
  # child is a folder somebody else may have named.
  while IFS= read -r line; do
    [[ $line == "$cur"* ]] && COMPREPLY+=("$(printf '%q' "$line")")
  done < <(command projekt folder list --output tsv --no-headers --short-only --verbose=error 2>/dev/null)
}

complete -F _pj_completion pj
