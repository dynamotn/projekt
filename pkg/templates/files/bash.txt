#!/bin/bash
# vim:filetype=sh
function pj() {
  if [ $# -ne 1 ]; then
    echo "usage: pj <project>" >&2
    return 1
  fi

  local target
  # `projekt folder get` exits non-zero when the short name is unknown, so the
  # error it already printed is the only message the user needs.
  target=$(command projekt folder get --verbose=error "$1") || return 1

  if [ ! -d "$target" ]; then
    echo "Project folder does not exist: $target" >&2
    return 1
  fi

  cd "$target" || return 1

  # "-" is the jump back, so the tab is named after where we landed.
  local label="$1"
  [ "$label" = "-" ] && label=$(basename "$target")
  [ -n "$ZELLIJ" ] && zellij action rename-tab "$label"
  [ -n "$TMUX" ] && tmux rename-window "$label"
  return 0
}

function _pj_completion() {
  local projects
  # --output tsv prints one short name per line, so completion does not have to
  # strip table borders and cannot be broken by a name containing one.
  projects=$(command projekt folder list --output tsv --no-headers --short-only --verbose=error 2>/dev/null)
  mapfile -t COMPREPLY < <(compgen -W "$projects" -- "${COMP_WORDS[COMP_CWORD]}")
}

complete -F _pj_completion pj
