# vim:filetype=fish
function pj --description "Jump to a project"
  if test (count $argv) -ne 1
    echo "usage: pj <project>" >&2
    return 1
  end
  # `projekt folder get` exits non-zero when the short name is unknown, so the
  # error it already printed is the only message the user needs.
  set -l target (command projekt folder get --verbose=error $argv[1])
  or return 1
  if not test -d "$target"
    echo "Project folder does not exist: $target" >&2
    return 1
  end
  cd $target
  or return 1
  test -n "$ZELLIJ"; and zellij action rename-tab $argv[1]
  test -n "$TMUX"; and tmux rename-window $argv[1]
end

function __pj_needs_command
  set cmd (commandline -opc)
  if [ (count $cmd) -eq 1 ]
    return 0
  else
    # Skip first word because it's "pj" or a wrapper
    for c in $cmd[2..-1]
      switch $c
        # We assume that any other token that's not an argument to a general option is a command
        case "*"
          return 1
      end
    end
    return 0
  end
  return 1
end

function __pj_completion
  command projekt folder list --no-headers --no-color --short-only --verbose=error \
    | command sed '1d;$d' \
    | string trim -c '|' \
    | string trim \
    | while read -l project
    test -n "$project"; or continue
    complete -f -c pj -n '__pj_needs_command' -a $project -d "Change to $project folder"
  end
end

function projekt
  command projekt $argv
  __pj_completion
end

__pj_completion
