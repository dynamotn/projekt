# vim:filetype=fish
function pj --description "Jump to a project"
  if test (count $argv) -gt 1
    echo "usage: pj [project]" >&2
    return 1
  end
  # `projekt folder get` exits non-zero when the short name is unknown, so the
  # error it already printed is the only message the user needs. With no
  # argument there is nothing to resolve, so ask instead.
  set -l target
  if test (count $argv) -eq 0
    set target (command projekt folder select --verbose=error)
    or return 1
  else
    set target (command projekt folder get --verbose=error $argv[1])
    or return 1
  end
  if not test -d "$target"
    echo "Project folder does not exist: $target" >&2
    return 1
  end
  cd $target
  or return 1

  # The argument is not the name we landed on when it was "-", a loose match
  # or nothing at all, so the tab is named after the folder itself.
  set -l label (basename $target)
  test -n "$ZELLIJ"; and zellij action rename-tab $label
  test -n "$TMUX"; and tmux rename-window $label
end

function __pj_needs_command
  set cmd (commandline -opc)
  if [ (count $cmd) -eq 1 ]
    return 0
  else
    # Skip first word because it's "pj" or a wrapper
    for c in $cmd[2..-1]
      switch $c
        # We assume that any other token that's not an argument to a general option is a command
        case "*"
          return 1
      end
    end
    return 0
  end
  return 1
end

function __pj_projects --description "Project short names, with their path as the description"
  # --output tsv prints one project per line, so completion does not have to
  # strip table borders and cannot be broken by a name containing one. The
  # first two columns are the short name and the path, which is what fish
  # wants: a candidate and its description, separated by a tab.
  command projekt folder list --output tsv --no-headers --verbose=error 2>/dev/null \
    | cut -f1,2
end

# The candidates are read when you press tab, not when this file is sourced, so
# a project created since — by `b`, by `projekt worktree add`, or by editing
# the config by hand — is offered without reloading anything.
complete -f -c pj -n '__pj_needs_command' -a '(__pj_projects)'
